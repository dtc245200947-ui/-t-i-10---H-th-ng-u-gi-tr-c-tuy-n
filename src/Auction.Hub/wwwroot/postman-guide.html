03 Chi tiết các Request GET/POST và Post-response Scripts

Request 01 — Health check (GET {{baseUrl}}/)

// Post-response Script:
pm.test("GET / trả về 200", function () {
  pm.response.to.have.status(200);
});
pm.test("Tên dịch vụ và trạng thái đúng", function () {
  const data = pm.response.json();
  pm.expect(data.service).to.equal("Auction.Api");
  pm.expect(data.status).to.equal("running");
});

Request 02 — Lấy danh sách phiên đấu giá (GET {{baseUrl}}/auctions)

// Post-response Script:
pm.test("GET /auctions trả về 200", function () {
  pm.response.to.have.status(200);
});
pm.test("Kết quả là mảng JSON các phiên đấu giá", function () {
  const data = pm.response.json();
  pm.expect(Array.isArray(data)).to.equal(true);
  pm.expect(data.length).to.be.greaterThan(0);
});

Request 03 — Tạo phiên đấu giá hợp lệ (POST {{baseUrl}}/auctions)

Chọn Body ➔ raw ➔ JSON:

{
  "productName": "  Laptop ASUS ROG Zephyrus G16 2026  ",
  "productDescription": "Chính hãng bảo hành 24 tháng",
  "startingPrice": 50000000,
  "minIncrement": 1000000,
  "durationMinutes": 60
}

Post-response Script (Kiểm tra 201 Created, Location Header và lưu auctionId):

pm.test("Tạo phiên đấu giá trả về 201 Created", function () {
  pm.response.to.have.status(201);
});
pm.test("JSON có dữ liệu phản hồi hợp lệ", function () {
  const data = pm.response.json();
  pm.expect(data.id).to.be.a("string");
  pm.expect(data.productName).to.equal("Laptop ASUS ROG Zephyrus G16 2026");
  pm.expect(data.startingPrice).to.equal(50000000);
  pm.expect(data.status).to.equal("Open");
});
pm.test("Location header chứa đường dẫn phiên đấu giá", () => {
  const location = pm.response.headers.get("Location");
  pm.expect(location).to.include("/auctions/");
});
const response = pm.response.json();
pm.collectionVariables.set("auctionId", response.id);

Request 07 — Đặt giá HỢP LỆ qua gRPC & RabbitMQ (POST {{baseUrl}}/auctions/{{auctionId}}/bids)

// Body (raw JSON):
{
  "bidderName": "Trần Đức Nhân",
  "amount": 52000000
}

// Post-response Script:
pm.test("Đặt giá hợp lệ trả về 200 OK", () => {
  pm.response.to.have.status(200);
});
pm.test("Phản hồi xác nhận thành công và có CorrelationId", () => {
  const data = pm.response.json();
  pm.expect(data.success).to.equal(true);
  pm.expect(data.currentPrice).to.equal(52000000);
  pm.expect(data.currentWinner).to.equal("Trần Đức Nhân");
  pm.expect(data.correlationId).to.be.a("string");
});

Request 08 — Đặt giá THẤP bị gRPC từ chối (POST {{baseUrl}}/auctions/{{auctionId}}/bids)

// Body (raw JSON - Giá 52.1tr thấp hơn mức tối thiểu 53tr):
{
  "bidderName": "Lê Hoàng Nam",
  "amount": 52100000
}

// Post-response Script:
pm.test("Giá đặt thấp hơn bước giá bị gRPC từ chối 400", () => {
  pm.response.to.have.status(400);
});
pm.test("Mã lỗi trả về từ gRPC là BID_TOO_LOW", () => {
  const data = pm.response.json();
  pm.expect(data.success).to.equal(false);
  pm.expect(data.errorCode).to.equal("BID_TOO_LOW");
});
